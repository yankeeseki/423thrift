<!--

  EDITING GUIDE FOR THE SHOP OWNER

  =================================

  1. MY RATINGS: each location has `myRating` (0-5) and `myNote` fields.

  2. FEATURED: set `featured: true` on any on-route location for the gold

     star treatment on the map and list.

  3. ADD / REMOVE: locations live in one combined `allLocations` list with a

     `defaultBonus` flag. Route-default locations are on the route

     automatically unless closed today or manually removed. Bonus-default

     locations sit off the route unless manually added.

  4. FEEDBACK EMAIL: built from character codes (search "EMAIL OBFUSCATION")

     rather than plain text, so it won't get scraped by simple bots. This is

     NOT true privacy \u2014 anyone reading the JavaScript can decode it. For a

     mailbox that never appears in the page at all, swap this for a

     form-backend service (e.g. Formspree).

  5. ROUTE ORDER: recalculated live in the browser (nearest-neighbor + 2-opt)

     whenever a location is added/removed or a store's closing time passes.

  6. PLAN-AHEAD SHARING: the "Share This Itinerary" link encodes the chosen

     date, time, and any add/remove overrides into the URL's query string.

     Anyone opening that link sees the exact same planned day restored

     automatically. There's no server-side database involved.

-->

  function parseHourToken(tok) {

    var parts = tok.split(':');

    return { h: parseInt(parts[0], 10), m: parts.length > 1 ? parseInt(parts[1], 10) : 0 };

  }

  function openValueTo24(tok) {

    var t = parseHourToken(tok);

    if (t.h === 12) return 12 + t.m / 60;

    if (t.h >= 7 && t.h <= 11) return t.h + t.m / 60;

    if (t.h >= 1 && t.h <= 6) return t.h + 12 + t.m / 60;

    return t.h + t.m / 60;

  }

  function closeValueTo24(tok) { var t = parseHourToken(tok); return t.h + 12 + t.m / 60; }

  function parseHoursString(str) {

    var schedule = [null, null, null, null, null, null, null];

    str.split(' &middot; ').forEach(function(seg) {

      seg = seg.trim();

      var spaceIdx = seg.indexOf(' ');

      var dayPart = seg.substring(0, spaceIdx);

      var restPart = seg.substring(spaceIdx + 1).trim();

      var days = [];

      if (dayPart === 'Daily') { days = [0,1,2,3,4,5,6]; }

      else if (dayPart.indexOf('-') !== -1) {

        var rp = dayPart.split('-');

        var s = DAY_ORDER.indexOf(rp[0]), e = DAY_ORDER.indexOf(rp[1]);

        for (var d = s; d <= e; d++) days.push(d);

      } else if (dayPart.indexOf('/') !== -1) {

        dayPart.split('/').forEach(function(d) { days.push(DAY_ORDER.indexOf(d)); });

      } else { days.push(DAY_ORDER.indexOf(dayPart)); }

      var isClosed = restPart === 'Closed';

      var oc = null;

      if (!isClosed) { var tp = restPart.split('-'); oc = [openValueTo24(tp[0]), closeValueTo24(tp[1])]; }

      days.forEach(function(d) { if (d >= 0) schedule[d] = isClosed ? null : oc; });

    });

    return schedule;

  }

  function getEasternNow() {

    var fmt = new Intl.DateTimeFormat('en-US', { timeZone: 'America/New_York', weekday: 'short', hour: 'numeric', minute: 'numeric', hour12: false });

    var weekday = 'Sun', hour = 0, minute = 0;

    fmt.formatToParts(new Date()).forEach(function(p) {

      if (p.type === 'weekday') weekday = p.value;

      if (p.type === 'hour') hour = parseInt(p.value, 10);

      if (p.type === 'minute') minute = parseInt(p.value, 10);

    });

    if (hour === 24) hour = 0;

    return { dayIdx: DAY_ORDER.indexOf(weekday), decHour: hour + minute / 60 };

  }

  function formatHour(dec) {

    var h12 = h % 12; if (h12 === 0) h12 = 12;

    return h12 + (m ? ':' + (m < 10 ? '0' + m : m) : '') + ' ' + period;

  }

  function haversineMiles(lat1, lon1, lat2, lon2) {

    var R = 3958.8;

    var toRad = function(d) { return d * Math.PI / 180; };

    var dLat = toRad(lat2 - lat1), dLon = toRad(lon2 - lon1);

    var a = Math.sin(dLat / 2) * Math.sin(dLat / 2) + Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLon / 2) * Math.sin(dLon / 2);

    return 2 * R * Math.asin(Math.sqrt(a));

  }

  function computeTightBounds(points) {

    if (points.length <= 4) return points.map(function(s) { return [s.lat, s.lng]; });

    var centerLat = points.reduce(function(sum, s) { return sum + s.lat; }, 0) / points.length;

    var centerLng = points.reduce(function(sum, s) { return sum + s.lng; }, 0) / points.length;

    var withDist = points.map(function(s) {

      return { s: s, d: haversineMiles(centerLat, centerLng, s.lat, s.lng) };

    });

    withDist.sort(function(a, b) { return a.d - b.d; });

    var keepCount = Math.max(Math.ceil(points.length * 0.85), 4);

    return withDist.slice(0, keepCount).map(function(x) { return [x.s.lat, x.s.lng]; });

  }


  function optimizeLoop(points, dayIdx, startDecHour) {

    var n = points.length;

    if (n <= 2) return points.slice();

    var dist = [];

    for (var i = 0; i < n; i++) { dist.push([]); for (var j = 0; j < n; j++) dist[i].push(haversineMiles(points[i].lat, points[i].lng, points[j].lat, points[j].lng)); }

    function routeLen(route) { var t = 0; for (var k = 0; k < route.length; k++) t += dist[route[k]][route[(k+1) % route.length]]; return t; }

    var unvisited = []; for (var u = 1; u < n; u++) unvisited.push(u);

    var route = [0], current = 0;

    while (unvisited.length) {

      var best = null, bestD = Infinity;

      unvisited.forEach(function(cand) { if (dist[current][cand] < bestD) { bestD = dist[current][cand]; best = cand; } });

      route.push(best); unvisited.splice(unvisited.indexOf(best), 1); current = best;

    }

    var improved = true;

    while (improved) {

      improved = false;

      for (var a = 1; a < n - 1; a++) {

        for (var b = a + 1; b < n; b++) {

          if (b - a === 1) continue;

          var newRoute = route.slice(0, a).concat(route.slice(a, b).reverse(), route.slice(b));

          if (routeLen(newRoute) < routeLen(route) - 1e-9) { route = newRoute; improved = true; }

        }

      }

    }


    // ---- Hours-aware refinement pass ----

    // The route above is purely geographic (shortest loop). If a dayIdx and

    // start time are given, do a second pass: try relocating individual

    // stops to a different position in the sequence (an "or-opt" move -

    // the right tool for "shift this one late-opening/late-closing stop

    // to a better slot", since a plain 2-opt reversal can't reposition a

    // single stop on its own). Only accept a relocation if it reduces

    // wasted waiting time or avoids arriving after close, AND the total

    // drive distance stays within a bounded budget (12% over the

    // geographically-optimal loop) - distance stays king; hours only get

    // to nudge order within that budget, never create a big detour.

    if (dayIdx !== undefined && startDecHour !== undefined) {

      var pureDistance = routeLen(route);

      var ASSUMED_MPH = 25, VISIT_MIN = 20;

      var maxDistance = pureDistance * 1.12;


      function evaluateSchedule(r) {

        var t = startDecHour, wait = 0, violations = 0;

        for (var k = 0; k < r.length; k++) {

          if (k > 0) { t += (dist[r[k - 1]][r[k]] / ASSUMED_MPH) * 60 / 60; }

          var sched = parseHoursString(points[r[k]].hours)[dayIdx];

          if (sched) {

            if (t < sched[0]) { wait += (sched[0] - t) * 60; t = sched[0]; }

            if (t >= sched[1]) violations++;

          }

          t += VISIT_MIN / 60;

        }

        return { wait: wait, violations: violations };

      }

      function combinedCost(r) {

        var e = evaluateSchedule(r);

        // wait time is dampened (a nudge, not a mandate); a shop you'd

        // arrive at after closing is weighted heavily since you simply

        // can't do that stop in that slot.

        return e.wait * 0.02 + e.violations * 20;

      }


      var currentCost = combinedCost(route);

      var orOptImproved = true;

      while (orOptImproved) {

        orOptImproved = false;

        for (var moveFrom = 1; moveFrom < route.length; moveFrom++) {

          var nodeToMove = route[moveFrom];

          var withoutNode = route.slice(0, moveFrom).concat(route.slice(moveFrom + 1));

          for (var moveTo = 1; moveTo <= withoutNode.length; moveTo++) {

            var candidate = withoutNode.slice(0, moveTo).concat([nodeToMove], withoutNode.slice(moveTo));

            if (routeLen(candidate) > maxDistance) continue;

            var candidateCost = combinedCost(candidate);

            if (candidateCost < currentCost - 1e-6) {

              route = candidate; currentCost = candidateCost; orOptImproved = true;

            }

          }

        }

      }

    }


    return route.map(function(idx) { return points[idx]; });

  }

        manualOverrides[name] = val === 'i' ? 'include' : 'skip';

        sharedOverridesApplied = true;

      }

    });

  }


  function isOnRouteToday(loc, now) {

    var override = manualOverrides[loc.name];

    if (override === 'include') return true;

    if (override === 'skip') return false;

    if (loc.defaultBonus) return false;

    var schedule = parseHoursString(loc.hours);

    var today = schedule[now.dayIdx];

    if (!today) return false;

    if (now.decHour >= today[1]) return false;

    return true;

  }

  function pickStartingLocation(uLat, uLng, excludeVisited) {

    var now = effectiveNow();

    var openNow = [], opensLater = [];

    allLocations.forEach(function(s, i) {

      if (!isOnRouteToday(s, now)) return;

      if (excludeVisited && visitedStores[s.name]) return;

      var sched = parseHoursString(s.hours)[now.dayIdx];

      if (!sched) return;

      var d = (uLat !== null) ? haversineMiles(uLat, uLng, s.lat, s.lng) : 0;

      if (now.decHour >= sched[0]) { openNow.push({ idx: i, dist: d }); }

      else { opensLater.push({ idx: i, dist: d, opensAt: sched[0] }); }

    });

    if (openNow.length) {

      openNow.sort(function(a, b) { return a.dist - b.dist; });

      return { idx: openNow[0].idx, reason: 'open' };

    }

    if (opensLater.length) {

      opensLater.sort(function(a, b) { return (a.opensAt - b.opensAt) || (a.dist - b.dist); });

      return { idx: opensLater[0].idx, reason: 'opens-soon', opensAt: opensLater[0].opensAt };

    }

    var bestIdx = 0, bestDist = Infinity;

    allLocations.forEach(function(s, i) {

      var d = (uLat !== null) ? haversineMiles(uLat, uLng, s.lat, s.lng) : i;

      if (d < bestDist) { bestDist = d; bestIdx = i; }

    });

    return { idx: bestIdx, reason: 'fallback' };

  }


  function computeTimerText() {

    var now = getEasternNow();

    var remaining = activeRoute.filter(function(s) { return !visitedStores[s.name]; });

    var latestClose = null, latestName = '', openTodayCount = 0;

    remaining.forEach(function(store) {

      var schedule = parseHoursString(store.hours);

      var today = schedule[now.dayIdx];

      if (today) { openTodayCount++; if (latestClose === null || today[1] > latestClose) { latestClose = today[1]; latestName = store.name; } }

    });

    // activeRoute already excludes anything that's closed for the day (see

    // isOnRouteToday in recalibrate), so an empty activeRoute means nothing

    // is open at all right now - a genuinely different situation from

    // "everything's open but you've visited it all", which is checked next.

    if (activeRoute.length === 0) return { text: 'All the stores are closed for the day &mdash; let\u2019s do it again tomorrow!', done: true, urgency: 'calm' };

    if (remaining.length === 0) return { text: 'Every shop on today\u2019s trail is marked visited \u2014 nice work!', done: true, urgency: 'calm' };

    if (openTodayCount === 0) return { text: 'Everything left on today\u2019s trail is closed &mdash; better luck tomorrow!', done: true, urgency: 'calm' };

    // Subtract 20 min off the raw countdown to closing - that's roughly

    // the minimum time needed to actually shop at that last stop, so

    // "time remaining" reflects when you'd need to arrive by, not the

    // literal closing time itself (which would leave zero time to browse).

    var LAST_STOP_MIN = 20;

    var minutesRemaining = Math.round((latestClose - now.decHour) * 60) - LAST_STOP_MIN;

    if (minutesRemaining <= 0) return { text: 'Everything left on today\u2019s trail has closed for the day. Time to call it!', done: true, urgency: 'calm' };

    var hrs = Math.floor(minutesRemaining / 60), mins = minutesRemaining % 60;

    var timeStr = (hrs > 0 ? hrs + 'h ' : '') + mins + 'm';


    // Simple average: remaining browsing time (after subtracting estimated

    // drive time) split evenly across every remaining shop.

    var driveMinutes = estimateRemainingDriveMinutes(remaining);

    var browsingMinutes = Math.max(minutesRemaining - driveMinutes, 0);

    var avgPerShop = Math.max(0, Math.floor(browsingMinutes / remaining.length));

    var fitNote = 'Apprx <strong>' + avgPerShop + ' minutes</strong> at each shop';


    // Urgency ramps up as real time actually runs out - genuine, since

    // shops really are closing, not a manufactured countdown gimmick.

    var urgency = 'calm', leadIn = '';

    if (minutesRemaining <= 30) { urgency = 'urgent'; leadIn = '&#9889; '; }

    else if (minutesRemaining <= 90) { urgency = 'tight'; leadIn = '&#9203; '; }


    return {

      text: '<div class="ctt-status-stat-row">' + leadIn + '<strong>' + timeStr + '</strong> remaining &middot; ' + fitNote + '</div>',

      done: false,

      urgency: urgency

    };

  }

  function estimateRemainingDriveMinutes(remaining) {

    if (remaining.length < 2) return 0;

    var ASSUMED_MPH = 25, total = 0;

    for (var i = 0; i < remaining.length - 1; i++) {

      total += haversineMiles(remaining[i].lat, remaining[i].lng, remaining[i + 1].lat, remaining[i + 1].lng) / ASSUMED_MPH * 60;

    }

    return Math.round(total);

  }

  // Resize + JPEG-compress a photo in the browser, then return it as a data

  // URL (a base64-encoded string). This gets stored directly inside the

  // Firestore review document - no Firebase Storage needed at all, which

  // means no Blaze plan / no card on file required. Firestore documents cap

  // out at 1MB total, so this shrinks the photo until it's safely under

  // that, trying progressively smaller sizes if needed.

  function resizeAndCompressImage(file, maxDim, quality) {

    return new Promise(function(resolve, reject) {

      var reader = new FileReader();

      reader.onload = function(e) {

        var img = new Image();

        img.onload = function() {

          var canvas = document.createElement('canvas');

          var w = img.width, h = img.height;

          if (w > h && w > maxDim) { h = Math.round(h * maxDim / w); w = maxDim; }

          else if (h >= w && h > maxDim) { w = Math.round(w * maxDim / h); h = maxDim; }

          canvas.width = w; canvas.height = h;

          var ctx = canvas.getContext('2d');

          ctx.drawImage(img, 0, 0, w, h);

          resolve(canvas.toDataURL('image/jpeg', quality));

        };

        img.onerror = function() { reject(new Error('image load failed')); };

        img.src = e.target.result;

      };

      reader.onerror = function() { reject(new Error('file read failed')); };

      reader.readAsDataURL(file);

    });

  }

  function compressUntilSmallEnough(file) {

    var attempts = [

      { maxDim: 1000, quality: 0.7 },

      { maxDim: 800, quality: 0.6 },

      { maxDim: 600, quality: 0.5 },

      { maxDim: 400, quality: 0.4 }

    ];

    var i = 0;

    function tryNext() {

      var opt = attempts[i];

      return resizeAndCompressImage(file, opt.maxDim, opt.quality).then(function(dataUrl) {

        if (dataUrl.length < 700000 || i === attempts.length - 1) return dataUrl;

        i++;

        return tryNext();

      });

    }

    return tryNext();

  }

  function renderBlendedStars(rating) {

    if (rating === null || rating === undefined) return '<span style="color:#b8ad95;">\u2606\u2606\u2606\u2606\u2606 no rating</span>';

    var full = Math.floor(rating);

    var half = rating % 1 >= 0.5;

    var str = '';

    for (var i = 0; i < full; i++) str += '\u2605';

    if (half) str += '\u00bd';

    var filledCount = full + (half ? 1 : 0);

    for (var j = filledCount; j < 5; j++) str += '\u2606';

    return '<span style="color:var(--gold);">' + str + '</span>';

  }
